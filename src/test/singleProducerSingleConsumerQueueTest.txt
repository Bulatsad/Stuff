#include <thread>
#include <iostream>
#include <mutex>
#include <time.h>

constexpr size_t runCount = 10000000;
constexpr size_t datasizeMB = ((double)(sizeof(size_t) * runCount) / (double)1000000);
static std::mutex mCout;

#include <blib/thread/circleQueue.h>

blib::LocklessProducerConcumerCircleQueue<size_t>cq(8);

void producer()
{
    size_t i = 0;
    size_t collisionCount = 0;
    while (i <= runCount)
    {
        while (!cq.push(i))
            collisionCount++;
        i++;
    }

    mCout.lock();
    std::cout << "producer collision : " << collisionCount << std::endl;
    mCout.unlock();

    return;
}

void consumer()
{
    
    size_t i = 0;
    size_t currentData;
    size_t collisionCount = 0;
    size_t errCount  = 0;

    clock_t start = clock();
    while (i <= runCount)
    {
        while (!cq.pop(currentData))
            collisionCount++;
        if (currentData != i)
        {
            if(errCount == 0)
                std::cout << "FAULT!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!" << std::endl;
            errCount++;
        }
        i++;
    }
    clock_t time = clock() - start;

    mCout.lock();
    std::cout << "consumer err count : " << errCount << std::endl;
    std::cout << "consumer collision : " << collisionCount << std::endl;
    std::cout << "speed : " << (double)datasizeMB / (double)(time / (double)1000)  << " MBytes/Sec" << std::endl;
    std::cout << "time : " << time << std::endl;
    mCout.unlock();

    return;
}

int main()
{
    std::thread t1(producer);
    std::thread t2(consumer);

    std::cout << "data size :" << datasizeMB << " MB" << std::endl;

    t1.join();
    t2.join();


    return 0;
}
